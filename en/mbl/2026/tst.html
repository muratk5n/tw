df = df.resample('YE').last()
df['PE'] = df['S&P Comp P'] / df['Earnings E']
df['Div Yield %'] = (df['Dividends D'] / df['S&P Comp P']) * 100
df['E Growth'] = df['Earnings E'].pct_change().rolling(window).mean() * 100
denom = df['E Growth'] + df['Div Yield %']
df['PEGY'] = df['PE'] / denom
